创建控制器

控制器位于 app/Controller 目录，继承自 Xqkeji\Controller 基类。

基础控制器

php namespace App\Controller;

use Xqkeji\Controller;

class UserController extends Controller
{
    public function index()
    {
        return $this->view('user/index');
    }
}

使用命令行创建

bash php xqkeji make:controller UserController

请求处理

通过 $this->request 获取请求对象。

获取请求参数

php // 获取所有参数
$data = $this->request()->all();

// 获取单个参数
$name = $this->request()->input('name');

// 获取带默认值
$page = $this->request()->input('page', 1);

// 判断参数是否存在
if ($this->request()->has('name')) {
    // ...
}

文件上传

php $file = $this->request()->file('avatar');
if ($file) {
    $path = $file->move('uploads/avatars');
}

响应返回

返回视图

php return $this->view('user/index', [
    'users' => $users
]);

返回 JSON

php return $this->json([
    'code' => 0,
    'data' => $data,
    'message' => 'success'
]);

重定向

php // 重定向到指定路由
return redirect('/user/list');

// 重定向并携带数据
return redirect('/user/list')->with('message', '操作成功');
